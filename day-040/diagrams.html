Python 实现建议

python
# ── 抽象基类方案（标准 OOP）──
from abc import ABC, abstractmethod

class Observer(ABC):
    @abstractmethod
    def update(self, data): ...

# ── Protocol 方案（鸭子类型）──
from typing import Protocol

class Observer(Protocol):
    def update(self, data): ...

# ── 函数式方案（轻量级）──
# 直接用 Callable[[Any], None] 替代观察者类
from typing import Callable

Observer = Callable[[Any], None]
