python
# 提示：使用 synonyms 库或维护同义词词典
# syn替换示例
import random

synonyms = {"好": ["优秀", "出色", "很棒"], "差": ["糟糕", "劣质", "不行"]}

def synonym_replace(text):
    for word, syns in synonyms.items():
        if word in text:
            text = text.replace(word, random.choice(syns), 1)
    return text
